#include <tunables/global>

# AppArmor profile of the app. The Supervisor loads it instead of Docker's
# default profile (docker-default) and names it after the app's full slug.
#
# Claude Code is a general tool: as root it reads and writes files anywhere in
# the container and in the mapped folders, runs any program and uses the
# network. So this profile allows all of that, keeps every restriction of
# docker-default (no mount, no writes to /proc and /sys, no ptrace of other
# containers) and allows only the capabilities listed below instead of all.
#
# complain: nothing is blocked by the rules below yet; what they would block
# is logged in the host journal as apparmor="ALLOWED". Remove complain once a
# test shows no such entries.
profile claude_code flags=(attach_disconnected,mediate_deleted,complain) {
  #include <abstractions/base>

  file,
  network,
  umount,

  # Capabilities root needs in the container: s6-overlay, bashio, apk (option
  # additional_packages), git, chown/chmod in the mapped folders, ping
  capability chown,
  capability dac_override,
  capability dac_read_search,
  capability fowner,
  capability fsetid,
  capability kill,
  capability net_bind_service,
  capability net_raw,
  capability setgid,
  capability setuid,

  # Signals: from the host (Supervisor, Docker) and within the app
  signal (receive) peer=unconfined,
  signal (receive) peer=runc,
  signal (receive) peer=crun,
  signal (send, receive) peer=@{profile_name},

  # ps, tmux (current path of a pane) and similar read /proc of the app's own
  # processes
  ptrace (trace, read, tracedby, readby) peer=@{profile_name},

  # Restrictions of docker-default
  deny @{PROC}/* w,
  deny @{PROC}/{[^1-9],[^1-9][^0-9],[^1-9s][^0-9y][^0-9s],[^1-9][^0-9][^0-9][^0-9/]*}/** w,
  deny @{PROC}/sys/[^k]** w,
  deny @{PROC}/sys/kernel/{?,??,[^s][^h][^m]**} w,
  deny @{PROC}/sysrq-trigger rwklx,
  deny @{PROC}/kcore rwklx,
  deny mount,
  deny /sys/[^f]*/** wklx,
  deny /sys/f[^s]*/** wklx,
  deny /sys/fs/[^c]*/** wklx,
  deny /sys/fs/c[^g]*/** wklx,
  deny /sys/fs/cg[^r]*/** wklx,
  deny /sys/firmware/** rwklx,
  deny /sys/devices/virtual/powercap/** rwklx,
  deny /sys/kernel/security/** rwklx,
}
